% Page styles — headers, footers
\usepackage{fancyhdr}
\usepackage{lastpage}

\pagestyle{fancy}
\fancyhf{}

% Book: chapter name in header, page number in footer
\fancyhead[LE,RO]{\small\thepage}
\fancyhead[LO]{\small\itshape\nouppercase{\rightmark}}
\fancyhead[RE]{\small\itshape\nouppercase{\leftmark}}
\fancyfoot{}

\renewcommand{\headrulewidth}{0.3pt}
\renewcommand{\footrulewidth}{0pt}

% Plain style for chapter opening pages
\fancypagestyle{plain}{
  \fancyhf{}
  \fancyfoot[C]{\small\thepage}
  \renewcommand{\headrulewidth}{0pt}
}

% Article footer: copyright line
\fancypagestyle{article}{
  \fancyhf{}
  \fancyhead[L]{\small\itshape\nouppercase{\rightmark}}
  \fancyhead[R]{\small\thepage}
  \fancyfoot[L]{%
    \scriptsize\textcolor{textsecondary}{%
      \ifdefined\PDFAuthor\PDFAuthor\fi
      \ifdefined\ProjectVersion\ \textbullet\ \ProjectVersion\fi
    }%
  }
  \renewcommand{\headrulewidth}{0.3pt}
  \renewcommand{\footrulewidth}{0.3pt}
}
